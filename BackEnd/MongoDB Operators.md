MongoDB Operators
│
├── Comparison
│   │
│   ├── $eq    → يساوي
│   ├── $ne    → لا يساوي
│   │
│   ├── $gt    → أكبر من
│   ├── $gte   → أكبر من أو يساوي
│   │
│   ├── $lt    → أصغر من
│   ├── $lte   → أصغر من أو يساوي
│   │
│   ├── $in    → واحد من مجموعة
│   └── $nin   → ليس من مجموعة
│
├── Logical
│   │
│   ├── $or    → أو
│   ├── $and   → و
│   ├── $nor   → لا هذا ولا هذا
│   └── $not   → عكس الشرط
│
├── Array
│   │
│   ├── $in        → يحتوي/يطابق واحدًا من
│   ├── $nin       → لا يحتوي/لا يطابق
│   ├── $all       → يحتوي على الكل
│   ├── $size      → عدد العناصر
│   └── $elemMatch → عنصر يحقق شروطًا
│
├── Element
│   │
│   ├── $exists → هل الـ field موجود؟
│   └── $type   → ما نوع الـ field؟
│
└── Evaluation
    │
    ├── $regex   → بحث بالنمط
    ├── $options → خيارات Regex
    └── $text    → Text Search
    
    
    /*
    
    
===========================================================
              MongoDB QUERY OPERATORS
===========================================================

الفكرة الأساسية:

MongoDB Query:
{
  field: operator: value
}

مثال:

{
  price: {
    $gt: 100
  }
}

اقرأها:
"هات الكتب التي سعرها أكبر من 100"

===========================================================
*/


/*
===========================================================
1️⃣ COMPARISON OPERATORS
   Operators للمقارنة
===========================================================

تستخدم عندما تريد مقارنة قيمة بقيمة أخرى.
*/


// $eq
// Equal
// يساوي
// غالبًا لا تحتاج كتابته، لأن MongoDB يفهم { price: 100 }

{
  price: {
    $eq: 100
  }
}


// نفس المعنى تقريبًا:
{
  price: 100
}


/*
-----------------------------------------------------------
$ne
Not Equal
لا يساوي
-----------------------------------------------------------
*/

{
  price: {
    $ne: 100
  }
}

// السعر ليس 100


/*
-----------------------------------------------------------
$gt
Greater Than
أكبر من
-----------------------------------------------------------
*/

{
  price: {
    $gt: 100
  }
}

// price > 100


/*
-----------------------------------------------------------
$gte
Greater Than or Equal
أكبر من أو يساوي
-----------------------------------------------------------
*/

{
  price: {
    $gte: 100
  }
}

// price >= 100


/*
-----------------------------------------------------------
$lt
Less Than
أصغر من
-----------------------------------------------------------
*/

{
  price: {
    $lt: 100
  }
}

// price < 100


/*
-----------------------------------------------------------
$lte
Less Than or Equal
أصغر من أو يساوي
-----------------------------------------------------------
*/

{
  price: {
    $lte: 100
  }
}

// price <= 100


/*
-----------------------------------------------------------
$in
داخل مجموعة من القيم
واحد من
-----------------------------------------------------------
*/

{
  price: {
    $in: [50, 100, 200]
  }
}

// السعر 50 أو 100 أو 200


/*
-----------------------------------------------------------
$nin
Not In
ليس داخل مجموعة من القيم
-----------------------------------------------------------
*/

{
  price: {
    $nin: [50, 100, 200]
  }
}

// السعر ليس 50 ولا 100 ولا 200



/*
===========================================================
2️⃣ LOGICAL OPERATORS
   Operators للمنطق
===========================================================
*/


/*
-----------------------------------------------------------
$or
أو
-----------------------------------------------------------

يكفي أن يتحقق شرط واحد.

مثال:
العنوان يحتوي على البحث
OR
المؤلف يطابق البحث
*/

{
  $or: [
    {
      title: {
        $regex: search,
        $options: "i"
      }
    },

    {
      author: {
        $in: authorIds
      }
    }
  ]
}


/*
-----------------------------------------------------------
$and
و
-----------------------------------------------------------

يجب أن تتحقق كل الشروط.
*/

{
  $and: [
    {
      price: {
        $gte: 100
      }
    },

    {
      price: {
        $lte: 500
      }
    }
  ]
}


/*
لكن غالبًا لا تحتاج $and هنا.
يمكنك كتابة:

*/

{
  price: {
    $gte: 100,
    $lte: 500
  }
}


/*
-----------------------------------------------------------
$nor
لا هذا ولا هذا
-----------------------------------------------------------
*/

{
  $nor: [
    {
      price: 100
    },

    {
      category: "Programming"
    }
  ]
}

// السعر ليس 100
// AND
// التصنيف ليس Programming


/*
-----------------------------------------------------------
$not
عكس الشرط
-----------------------------------------------------------
*/

{
  price: {
    $not: {
      $gt: 100
    }
  }
}

// السعر ليس أكبر من 100



/*
===========================================================
3️⃣ ARRAY OPERATORS
   مهمة جدًا عندك لأن categories عبارة عن Array
===========================================================
*/


/*
-----------------------------------------------------------
$in
القيمة تطابق واحدة من القيم
-----------------------------------------------------------
*/

{
  categories: {
    $in: [categoryId1, categoryId2]
  }
}

// الكتاب عنده واحد على الأقل من هذه التصنيفات


/*
-----------------------------------------------------------
$nin
القيمة لا تطابق أيًا من القيم
-----------------------------------------------------------
*/

{
  categories: {
    $nin: [categoryId1, categoryId2]
  }
}

// الكتاب ليس ضمن هذه التصنيفات


/*
-----------------------------------------------------------
$all
يجب أن يحتوي Array على كل القيم
-----------------------------------------------------------
*/

{
  categories: {
    $all: [
      categoryId1,
      categoryId2
    ]
  }
}

// الكتاب يجب أن يحتوي على التصنيفين معًا


/*
-----------------------------------------------------------
$size
عدد عناصر الـ Array
-----------------------------------------------------------
*/

{
  categories: {
    $size: 2
  }
}

// الكتاب لديه بالضبط تصنيفين



/*
===========================================================
4️⃣ ELEMENT OPERATORS
   هل الـ Field موجود؟ وما نوعه؟
===========================================================
*/


/*
-----------------------------------------------------------
$exists
هل الـ field موجود؟
-----------------------------------------------------------
*/

{
  publisher: {
    $exists: true
  }
}

// publisher موجود


{
  publisher: {
    $exists: false
  }
}

// publisher غير موجود


/*
-----------------------------------------------------------
$type
نوع البيانات
-----------------------------------------------------------
*/

{
  price: {
    $type: "number"
  }
}

// price نوعه Number



/*
===========================================================
5️⃣ EVALUATION OPERATORS
   Operators لتقييم النصوص والقيم
===========================================================
*/


/*
-----------------------------------------------------------
$regex
البحث باستخدام Pattern
-----------------------------------------------------------
*/

{
  title: {
    $regex: "javascript"
  }
}

// يبحث عن javascript داخل title


/*
-----------------------------------------------------------
$options
خيارات الـ Regex
-----------------------------------------------------------
*/

{
  title: {
    $regex: "javascript",
    $options: "i"
  }
}

// i = Case Insensitive
// لا يفرق بين:
// JavaScript
// javascript
// JAVASCRIPT


/*
-----------------------------------------------------------
Regex جزئي
-----------------------------------------------------------

لو search = "java"

سيجد:

JavaScript
Java
Learn Java
*/

{
  title: {
    $regex: search,
    $options: "i"
  }
}


/*
===========================================================
6️⃣ ARRAY ELEMENT MATCH
===========================================================
*/


/*
-----------------------------------------------------------
$elemMatch
البحث داخل Array بحيث تحقق العناصر شروطًا معينة.
-----------------------------------------------------------
*/

/*
مثال افتراضي:

ratings: [
  {
    user: ...,
    rating: 5
  }
]

نريد Rating أكبر من 4
*/

{
  ratings: {
    $elemMatch: {
      rating: {
        $gt: 4
      }
    }
  }
}

  }

/*

🔟 COMBINING OPERATORS
===========================================================

يمكنك تركيب أكثر من Operator مع بعض.
===========================================================
*/


/*
السعر بين 100 و500
*/

{
  price: {
    $gte: 100,
    $lte: 500
  }
}


/*
السعر بين 100 و500
AND
الكتاب غير محذوف
*/

{
  price: {
    $gte: 100,
    $lte: 500
  },

  isDeleted: false
}


/*
العنوان يحتوي JavaScript
OR
المؤلف من مجموعة معينة
*/

{
  $or: [
    {
      title: {
        $regex: "javascript",
        $options: "i"
      }
    },

    {
      author: {
        $in: authorIds
      }
    }
  ]
}


/*
العنوان أو المؤلف
AND
التصنيف
AND
السعر بين 100 و500
AND
الكتاب غير محذوف
*/

{
  $or: [
    {
      title: {
        $regex: search,
        $options: "i"
      }
    },

    {
      author: {
        $in: authorIds
      }
    }
  ],
